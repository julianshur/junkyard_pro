import { ensureSchema, replaceRows } from "./db.js"; import { scrapePypInventory, FLAT_RATES, CORE_DEPOSITS } from "./pyp.js"; import { categorize } from "./match.js"; import { launchBrowser, scrapeEbaySold } from "./ebay.js"; // eBay fee + shipping haircut applied to the sold price before profit. // Default 0 to preserve your current "median - yardPrice" math. // Set FEE_RATE=0.15 (env) for a more realistic net. const FEE_RATE = parseFloat(process.env.FEE_RATE || "0"); async function run() { const t0 = Date.now(); console.log(`[${new Date().toISOString()}] run start`); await ensureSchema(); // 1. PYP inventory (free, direct axios). const inventory = await scrapePypInventory(); console.log(` PYP parts: ${inventory.length}`); if (!inventory.length) { console.log(" no inventory — leaving previous data in place, exiting."); return; } // 2. Categorize unique part names (Haiku, cached). One eBay query per category. const queryByPart = new Map(); for (const p of inventory) { if (!queryByPart.has(p.partName)) { queryByPart.set(p.partName, await categorize(p.partName)); } } const uniqueQueries = [...new Set([...queryByPart.values()].map((c) => c.ebayQuery))]; console.log(` unique eBay queries: ${uniqueQueries.length}`); // 3. eBay sold comps, sequential through ONE browser (polite to your IP). const browser = await launchBrowser(); const compByQuery = new Map(); try { let i = 0; for (const q of uniqueQueries) { compByQuery.set(q, await scrapeEbaySold(browser, q)); if (++i % 10 === 0) console.log(` comps ${i}/${uniqueQueries.length}`); } } finally { await browser.close(); } // 4. Join inventory -> comps, compute profit. const now = new Date().toISOString(); const rows = []; for (const p of inventory) { const { category, ebayQuery } = queryByPart.get(p.partName); if (!category) continue; // Haiku couldn't map it to a priced bucket -> skip const comp = compByQuery.get(ebayQuery); if (!comp) continue; // no comps -> not a profit opportunity worth showing // PYP flat rate for the bucket, plus the (non-refundable, for a flipper) // core deposit where one applies. const yardPrice = round2(FLAT_RATES[category] + (CORE_DEPOSITS[category] || 0)); const net = comp.median * (1 - FEE_RATE); const profit = round2(net - yardPrice); rows.push({ yard: p.yard, yardSlug: p.yardSlug, partName: p.partName, category, yardPrice, ebayMedian: comp.median, ebayLow: comp.low, ebayHigh: comp.high, ebaySoldCount: comp.count, ebayUrl: comp.url, profit, margin: yardPrice ? round2(profit / yardPrice) : null, updatedAt: now, }); } // 5. Atomic swap into Turso. await replaceRows(rows); console.log(` wrote ${rows.length} rows in ${((Date.now() - t0) / 1000).toFixed(1)}s`); } const round2 = (n) => Math.round(n * 100) / 100; run().catch((e) => { console.error("run failed:", e); process.exit(1); // non-zero so Task Scheduler can flag failures });
